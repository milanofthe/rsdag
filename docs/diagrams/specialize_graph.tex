% A piecewise model at v = 1, on and in the linear region: the arms taken
% and the conditions that guard them stay, the faded arms leave the
% specialized tape. A select is drawn as a multiplexer: its arms enter at
% the top (1 then, 0 else), its condition at the side.
\documentclass[tikz,border=4pt]{standalone}
\input{style}
% \mux{name}{x}{y}: a select centered at (x, y) with the anchors name-1,
% name-0 (the arms), name-l, name-r (a condition from either side) and
% name-o (the result).
\newcommand\mux[3]{
  \draw[accent] (#2 - 0.85, #3 + 0.35) -- (#2 + 0.85, #3 + 0.35)
    -- (#2 + 0.5, #3 - 0.35) -- (#2 - 0.5, #3 - 0.35) -- cycle;
  \coordinate (#1-1) at (#2 - 0.6, #3 + 0.35);
  \coordinate (#1-0) at (#2 + 0.6, #3 + 0.35);
  \coordinate (#1-l) at (#2 - 0.675, #3);
  \coordinate (#1-r) at (#2 + 0.675, #3);
  \coordinate (#1-o) at (#2, #3 - 0.35);
  \node[lbl, accent, font=\scriptsize] at (#2 - 0.6, #3 + 0.12) {1};
  \node[lbl, accent, font=\scriptsize] at (#2 + 0.6, #3 + 0.12) {0};
}
\begin{document}
\begin{tikzpicture}
  \node[input] (v) at (1.3, 0) {$v$};
  \node[input] (vth) at (2.7, 0) {$v_{th}$};
  \node[input] (vsat) at (4.6, 0) {$v_{sat}$};
  \node[op] (d) at (2.0, -1.3) {$-$};
  \node[op, accent] (c2) at (4.6, -1.3) {$>$};
  \node[op, accent] (c1) at (0.5, -2.7) {$> 0$};
  \node[op] (kd) at (2.0, -2.7) {$k\,d$};
  \node[op, faded] (half) at (3.2, -2.7) {$\frac{k}{2} d^2$};
  \mux{inner}{2.6}{-4.0}
  \node[op, faded, minimum size=6mm] (zero) at (3.8, -4.75) {$0$};
  \mux{outer}{3.2}{-5.8}
  \node[output] (i) at (3.2, -7.0) {$i$};

  \draw[data] (v) -- (d);
  \draw[data] (vth) -- (d);
  \draw[data] (vsat) -- (c2);
  \draw[data] (d) -- node[lbl, above] {$d$} (c2);
  \draw[data] (d) -- (c1);
  \draw[data] (d) -- (kd);
  \draw[data, faded] (d) -- (half);
  \draw[data] (kd) -- (inner-1);
  \draw[data, faded] (half) -- (inner-0);
  \draw[data, accent] (c2) |- (inner-r);
  \draw[data] (inner-o) -- (outer-1);
  \draw[data, faded] (zero) -- (outer-0);
  \draw[data, accent] (c1) |- (outer-l);
  \draw[data] (outer-o) -- (i);
\end{tikzpicture}
\end{document}
